-- Create login table
CREATE TABLE IF NOT EXISTS `mpass` (
    `id` INT(11) NOT NULL AUTO_INCREMENT,
    `user` VARCHAR(50) NOT NULL,
    `mpass` VARCHAR(50) NOT NULL,
    `role` INT(1) NOT NULL DEFAULT 1,
    `active` INT(1) DEFAULT 1,
    `dateent` DATETIME DEFAULT NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_user` (`user`)
) ENGINE=MyISAM;

-- Create gledg if not exists
CREATE TABLE IF NOT EXISTS `gledg` (
    `id` INT(11) NOT NULL AUTO_INCREMENT,
    `acc_code` VARCHAR(20) NOT NULL,
    `date` DATE NOT NULL,
    `amt_type` VARCHAR(5) NOT NULL,
    `amount` DECIMAL(10,2) NOT NULL,
    `description` VARCHAR(255) DEFAULT NULL,
    `bill_no` INT(11) DEFAULT NULL,
    `user` VARCHAR(50) DEFAULT NULL,
    `dateent` DATETIME DEFAULT NULL,
    PRIMARY KEY (`id`),
    INDEX `idx_acc_code` (`acc_code`),
    INDEX `idx_date` (`date`),
    INDEX `idx_bill_no` (`bill_no`)
) ENGINE=MyISAM;

When payment is received:

Cash → gledg.amt_type = 'CR', acc_code = '112000001'
Bank/Card/Easypaisa → gledg.amt_type = 'BR', acc_code = '112000001'
Admin refund → Creates reverse entry with negative amount
Admin delete → Logs as ADJ (Adjustment) type
